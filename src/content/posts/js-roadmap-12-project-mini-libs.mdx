> [!NOTE]
> **为什么 JS 的毕业项目是"手写库"而不是"做个页面"**：
> 做页面练的是组织代码，手写库练的是**语言机制本身**——闭包怎么保存状态、`this` 怎么丢、Promise 为什么是微任务、
> 监听器为什么回收不掉。这些是"能不能独自排查问题"的分水岭。
>
> **这个项目零依赖、可自动验证**：每个库都配 `node:test` 断言，写完跑 `node --test` 就知道对不对——
> 不用等别人 review。

5. 能写出真实的无限滚动：`IntersectionObserver` + 分页 + 图片懒加载，滚到底不卡。
6. 能在 DevTools 的 Memory 面板里**证明**页面没有泄漏：反复进入/离开后 detached 节点不增长。

## 阶段 1：三个同步原语（闭包与 this 的实战）

### 1. `EventEmitter`：观察者模式的最小内核

```js
// src/event-emitter.js
export class EventEmitter {
	#listeners = new Map(); // eventName -> Set<{ fn, once }>

	on(name, fn) {
		if (typeof fn !== "function") throw new TypeError("listener must be a function");
		if (!this.#listeners.has(name)) this.#listeners.set(name, new Set());
		this.#listeners.get(name).add({ fn, once: false });
		return () => this.off(name, fn); // 返回取消函数，调用方不用记住 name
	}

	once(name, fn) {
		const set = this.#listeners.get(name) ?? new Set();
		this.#listeners.set(name, set);
		const wrapper = (...args) => {
			this.off(name, wrapper);
			fn(...args);
		};
		// 关键：把原函数挂在 wrapper 上，off(name, 原函数) 才能删掉它
		wrapper.__origin = fn;
		set.add({ fn: wrapper, once: true });
		return () => this.off(name, fn);
	}

	off(name, fn) {
		const set = this.#listeners.get(name);
		if (!set) return;
		for (const record of set) {
			if (record.fn === fn || record.fn.__origin === fn) set.delete(record);
		}
		if (set.size === 0) this.#listeners.delete(name);
	}

	emit(name, ...args) {
		const set = this.#listeners.get(name);
		if (!set) return false;
		// 复制一份再遍历：监听器里 off/on 不会影响本次派发
		for (const record of [...set]) {
			try {
				record.fn(...args);
			} catch (err) {
				// 一个监听器报错不能拖垮其它监听器，交给全局错误处理
				console.error(`[EventEmitter] listener error on "${name}"`, err);
			}
		}
		return true;
	}

	removeAll(name) {
		if (name === undefined) this.#listeners.clear();
		else this.#listeners.delete(name);
	}

	get count() {
		let n = 0;
		for (const set of this.#listeners.values()) n += set.size;
		return n;
	}
}
```

**为什么不用 `Set<Function>` 存**：`once` 需要"用包装函数监听、但用原函数取消"。存一个带 `once` 标记的对象，才能让 `off(name, 原函数)` 也删得掉。这就是"闭包 + 引用关系"的实际用途，不是面试题。

### 2. `debounce`：只关心"停下来之后"

```js
// src/debounce.js
export function debounce(fn, wait = 300, { leading = false } = {}) {
	let timer = null;
	let pending = null; // { ctx, args }
	let leadingDone = false;

	const run = () => {
		timer = null;
		leadingDone = false;
		const p = pending;
		pending = null;
		if (p) fn.apply(p.ctx, p.args);
	};

	function debounced(...args) {
		pending = { ctx: this, args };
		const isFirst = timer === null;
		if (timer) clearTimeout(timer);
		timer = setTimeout(run, wait);

		// leading：窗口内的第一次立刻执行，剩下的仍然等安静下来补最后一次
		if (leading && isFirst && !leadingDone) {
			leadingDone = true;
			const p = pending;
			pending = null;
			fn.apply(p.ctx, p.args);
		}
	}

	debounced.cancel = () => {
		if (timer) clearTimeout(timer);
		timer = null;
		pending = null;
		leadingDone = false;
	};

	debounced.flush = () => {
		if (timer) clearTimeout(timer);
		timer = null;
		leadingDone = false;
		const p = pending;
		pending = null;
		if (p) fn.apply(p.ctx, p.args);
	};

	return debounced;
}
```

**`cancel` 不是可选项**：组件卸载、请求被新的取代、路由切换——不 cancel 就会在"已经不需要"的场景里执行回调，这是最典型的隐蔽 bug。

### 3. `throttle`：保证"最多每 N 毫秒一次"，且首尾都不丢

```js
// src/throttle.js
export function throttle(fn, interval = 200, { leading = true, trailing = true } = {}) {
	let lastTime = 0;
	let timer = null;
	let lastArgs = null;

	const invoke = (args) => {
		lastTime = Date.now();
		fn.apply(null, args);
	};

	function throttled(...args) {
		const now = Date.now();
		if (!lastTime && !leading) lastTime = now;

		const remaining = interval - (now - lastTime);
		lastArgs = args;

		if (remaining <= 0) {
			if (timer) {
				clearTimeout(timer);
				timer = null;
			}
			invoke(args);
			lastArgs = null;
			return;
		}

		// trailing：把这段时间内的最后一次调用补上
		if (trailing && !timer) {
			timer = setTimeout(() => {
				timer = null;
				lastTime = leading ? Date.now() : 0;
				if (lastArgs) {
					invoke(lastArgs);
					lastArgs = null;
				}
			}, remaining);
		}
	}

	throttled.cancel = () => {
		if (timer) clearTimeout(timer);
		timer = null;
		lastTime = 0;
		lastArgs = null;
	};

	return throttled;
}
```

**debounce 与 throttle 的选择**：搜索框输入用 debounce（只关心最终值）；滚动/拖拽/resize 用 throttle（需要过程中的连续反馈，但不能每像素都算）。

## 阶段 2：异步——手写 Promise、并发池、重试

### 1. `MiniPromise`：把"状态机"写出来

```js
// src/mini-promise.js
const PENDING = 0;
const FULFILLED = 1;
const REJECTED = 2;

export class MiniPromise {
	#state = PENDING;
	#value = undefined;
	#handlers = []; // { onFulfilled, onRejected, resolve, reject }

	constructor(executor) {
		const settle = (state, value) => {
			if (this.#state !== PENDING) return; // 只能 settle 一次
			this.#state = state;
			this.#value = value;
			// 微任务里统一处理：保证 then 的回调永远晚于当前同步代码
			queueMicrotask(() => {
				for (const h of this.#handlers.splice(0)) this.#run(h);
			});
		};

		try {
			executor(
				(value) => settle(FULFILLED, value),
				(reason) => settle(REJECTED, reason),
			);
		} catch (err) {
			settle(REJECTED, err);
		}
	}

	#run(handler) {
		const { onFulfilled, onRejected, resolve, reject } = handler;
		const cb = this.#state === FULFILLED ? onFulfilled : onRejected;
		if (typeof cb !== "function") {
			// 没传对应回调就"透传"当前状态
			this.#state === FULFILLED ? resolve(this.#value) : reject(this.#value);
			return;
		}
		try {
			resolve(cb(this.#value));
		} catch (err) {
			reject(err);
		}
	}

	then(onFulfilled, onRejected) {
		return new MiniPromise((resolve, reject) => {
			const handler = { onFulfilled, onRejected, resolve, reject };
			if (this.#state === PENDING) this.#handlers.push(handler);
			else queueMicrotask(() => this.#run(handler));
		});
	}

	catch(onRejected) {
		return this.then(undefined, onRejected);
	}

	finally(onSettled) {
		return this.then(
			(v) => {
				onSettled();
				return v;
			},
			(e) => {
				onSettled();
				throw e;
			},
		);
	}

	static resolve(value) {
		return value instanceof MiniPromise ? value : new MiniPromise((r) => r(value));
	}

	static reject(reason) {
		return new MiniPromise((_, r) => r(reason));
	}

	static all(list) {
		return new MiniPromise((resolve, reject) => {
			const results = new Array(list.length);
			let remaining = list.length;
			if (remaining === 0) return resolve(results);
			list.forEach((item, i) => {
				MiniPromise.resolve(item).then((v) => {
					results[i] = v;
					if (--remaining === 0) resolve(results);
				}, reject);
			});
		});
	}
}
```

**三个必须能解释的点**：

- `settle` 只能生效一次（`if (this.#state !== PENDING) return`）——这就是 `resolve` 之后再 `reject` 无效的原因。
- 用 `queueMicrotask` 而不是 `setTimeout(fn, 0)`——前者进微任务队列，与原生 Promise 顺序一致。
- `then` 没传回调要**透传**，否则 `promise.catch` 断链。

### 2. `asyncPool`：把并发压到 N

```js
// src/async-pool.js
/**
 * @param {number} limit 并发上限
 * @param {Array<() => Promise<any>>} tasks 任务工厂（不要传已经启动的 Promise！）
 */
export async function asyncPool(limit, tasks) {
	const results = new Array(tasks.length);
	let cursor = 0;

	const worker = async () => {
		while (cursor < tasks.length) {
			const index = cursor++;
			results[index] = await tasks[index]();
		}
	};

	await Promise.all(Array.from({ length: Math.min(limit, tasks.length) }, worker));
	return results;
}
```

**为什么传"工厂"而不是 Promise**：`tasks.map(fetch)` 一写出来，请求就已经全部发出去了，`limit` 形同虚设。这是最常见的误用。

### 3. `retry`：指数退避 + 只重试该重试的

```js
// src/retry.js
export async function retry(fn, { times = 3, delay = 300, factor = 2, shouldRetry = () => true } = {}) {
	let attempt = 0;
	for (;;) {
		try {
			return await fn(attempt);
		} catch (err) {
			attempt++;
			if (attempt >= times || !shouldRetry(err)) throw err;
			await new Promise((r) => setTimeout(r, delay * factor ** (attempt - 1)));
		}
	}
}
```

`shouldRetry` 是重点：**401 重试没有意义，503 才有**。生产代码里这个判断往往比重试本身更重要。

### 4. 用测试把顺序与并发钉住

```js
// test/async.test.js
import assert from "node:assert/strict";
import test from "node:test";
import { MiniPromise } from "../src/mini-promise.js";
import { asyncPool } from "../src/async-pool.js";
import { retry } from "../src/retry.js";

test("MiniPromise 回调晚于同步代码", async () => {
	const order = [];
	MiniPromise.resolve(1).then(() => order.push("then"));
	order.push("sync");
	await new Promise((r) => setTimeout(r, 0));
	assert.deepEqual(order, ["sync", "then"]);
});

test("MiniPromise.all 保序 + 一个失败整体失败", async () => {
	const value = await MiniPromise.all([
		MiniPromise.resolve(1),
		new MiniPromise((r) => setTimeout(() => r(2), 5)),
	]);
	assert.deepEqual(value, [1, 2]);
	await assert.rejects(
		MiniPromise.all([MiniPromise.resolve(1), MiniPromise.reject(new Error("x"))]),
		/x/,
	);
});

test("asyncPool 并发不超过上限", async () => {
	let running = 0;
	let peak = 0;
	const make = () => async () => {
		running++;
		peak = Math.max(peak, running);
		await new Promise((r) => setTimeout(r, 5));
		running--;
		return 1;
	};
	const tasks = Array.from({ length: 10 }, make);
	await asyncPool(3, tasks);
	assert.equal(peak, 3);
});

test("retry 遇到不该重试的错误直接抛", async () => {
	let calls = 0;
	await assert.rejects(
		retry(
			async () => {
				calls++;
				throw Object.assign(new Error("unauthorized"), { status: 401 });
			},
			{ times: 3, delay: 1, shouldRetry: (e) => e.status >= 500 },
		),
		/unauthorized/,
	);
	assert.equal(calls, 1);
});
```

```js
// web/gallery.js
import { throttle } from "../src/throttle.js";

const PAGE_SIZE = 12;
const MAX_PAGES = 6;

const grid = document.querySelector("#grid");
const status = document.querySelector("#status");
const sentinel = document.querySelector("#sentinel");

let page = 0;
let loading = false;
let done = false;
let itemCount = 0;
let observer = null;

/** 假接口：真实项目换 fetch，并把这个函数交给 retry() 包一层 */
async function fetchPage(nextPage) {
	await new Promise((r) => setTimeout(r, 400));
	return Array.from({ length: PAGE_SIZE }, (_, i) => {
		const id = (nextPage - 1) * PAGE_SIZE + i + 1;
		return { id, url: `https://picsum.photos/seed/${id}/480/360` };
	});
}

function render(items) {
	const frag = document.createDocumentFragment();
	for (const item of items) {
		const cell = document.createElement("div");
		cell.className = "cell";
		const img = document.createElement("img");
		img.alt = `图片 ${item.id}`;
		img.loading = "lazy";
		img.decoding = "async";
		img.dataset.src = item.url; // 先不写 src，交给 IntersectionObserver
		img.addEventListener("load", () => img.classList.add("loaded"), { once: true });
		cell.append(img);
		frag.append(cell);
	}
	grid.append(frag);
	itemCount += items.length;
	status.textContent = `已加载 ${itemCount} 张`;
}

/** 图片进入视口才真正请求，省流量也省首屏时间 */
const imageObserver = new IntersectionObserver(
	(entries) => {
		for (const entry of entries) {
			if (!entry.isIntersecting) continue;
			const img = entry.target;
			if (img.dataset.src && !img.src) img.src = img.dataset.src;
			imageObserver.unobserve(img);
		}
	},
	{ rootMargin: "200px" },
);

async function loadMore() {
	if (loading || done) return;
	if (page >= MAX_PAGES) {
		done = true;
		status.textContent = `没有更多了（共 ${itemCount} 张）`;
		observer?.disconnect();
		return;
	}

	loading = true;
	status.textContent = "加载中…";
	try {
		page++;
		const items = await fetchPage(page);
		render(items);
		grid.querySelectorAll("img:not([src])").forEach((img) => imageObserver.observe(img));
	} catch (err) {
		status.textContent = `加载失败：${err.message}`;
	} finally {
		loading = false;
	}
}

// 哨兵进入视口 → 加载下一页
observer = new IntersectionObserver(
	(entries) => {
		if (entries.some((e) => e.isIntersecting)) void loadMore();
	},
	{ rootMargin: "300px" },
);
observer.observe(sentinel);

// 滚动时只做"轻活"：读位置、更新进度条之类。用 throttle 保证 16ms 内最多一次
const onScrollLight = throttle(() => {
	document.body.dataset.scrolled = String(window.scrollY > 0);
}, 100);
window.addEventListener("scroll", onScrollLight, { passive: true });

// ===== 场景：这个模块要被"卸载"（单页应用切路由 / 组件销毁）=====
window.addEventListener("pagehide", () => {
	observer?.disconnect();
	imageObserver.disconnect();
	onScrollLight.cancel();
	window.removeEventListener("scroll", onScrollLight);
	grid.replaceChildren(); // 主动释放 DOM，避免 detached 节点堆积
});
```

### 泄漏实验（这个项目最有价值的一步）

1. 打开页面，滚到底加载 3 页。
2. DevTools → Memory → 拍一张 Heap snapshot，筛选 `Detached`，记下数量。
3. **故意**把 `pagehide` 里的清理代码注释掉，重复"进入 → 滚动 → 离开"5 次。
4. 再拍快照：detached 节点数量随次数线性增长 → 这就是泄漏。
5. 恢复清理代码，重复 5 次，数量**回到基线不增长**。

> 只看代码"觉得"没有泄漏不算数，**要能给出两张快照的对比数字**。这是"能独自排查"的最硬证据。

## 常见坑

1. **`once` 直接 `on(name, fn)`**：只执行一次但监听器还挂在 Map 里，频繁触发的事件（resize/mousemove）会越积越多。
2. **`emit` 里不复制数组**：监听器在回调里 `off` 自己，会导致遍历跳过下一个监听器。
3. **debounce 不 cancel**：组件卸载/请求被取代后仍执行，赋值给已销毁的状态。
4. **throttle 只做 `lastTime` 时间戳**：最后一次调用会被丢掉（没有 trailing），拖拽结束位置不准。
5. **`tasks.map(fetch)` 再用 `asyncPool`**：请求已经全发出去了，限流形同虚设。
6. **`MiniPromise` 用 `setTimeout` 代替 `queueMicrotask`**：顺序与原生不一致，踩坑时找不到原因。
7. **图片只写 `loading="lazy"` 就以为够了**：首屏图片被 lazy 反而拖慢 LCP，首屏该用 `fetchpriority="high"`。
8. **`pagehide` 里只 `disconnect` Observer，忘了移除 scroll 监听**：监听器持有 DOM 引用，照样泄漏。

## 毕业自测：8 题

1. `EventEmitter.once` 为什么必须包一层？`off` 怎么找到那个包装函数？
2. `debounce` 的 `flush()` 在什么场景下必须用？（提交前强制吐出最后一次输入）
3. `throttle` 的 `leading: false, trailing: true` 适合什么交互？（滚动到底部时才更新指示器）
4. `MiniPromise` 里 `settle` 的幂等判断去掉会怎样？（`resolve` 后 `reject` 也能生效，状态被改写）
5. `queueMicrotask` 与 `Promise.then` 谁先执行？（按入队顺序，同一队列）
6. `asyncPool` 的 `cursor++` 为什么不需要加锁？（JS 单线程，`await` 之前的同步段不会被打断）
7. 图片墙为什么用"哨兵 + IntersectionObserver"而不是监听 scroll 算位置？（不触发同步布局，回调在渲染后批量执行）
8. 你怎么证明页面没有内存泄漏？（Heap snapshot 对比 detached 节点数量，给出前后数字）
