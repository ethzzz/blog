## 做到什么算过

1. 能用 `net/http` + `ServeMux` 写出 `GET /tasks/{id}` 级别的路由，不需要框架。
2. 分页、筛选、搜索的入参解析有边界保护（负数、超上限、非数字都不会 500）。
3. 错误统一：404 / 422 / 500 三种情形返回同一结构的 JSON，且带 `requestId`。
4. 数据层从内存换到 SQLite，**业务层代码不用改**（靠接口隔离）。
5. `go test ./... -race -cover` 全绿，覆盖率 ≥ 60%，且错误分支有测试。
6. 用 `CGO_ENABLED=0 go build` 出一个静态二进制，扔到服务器能直接跑。
7. 优雅关闭：Ctrl+C 后在途请求不丢、进程 1 秒内退出。
8. 前端（React 那篇）把 base 指过来，列表/新建/删除全部可用，**前端零改动**。

## 契约（与另外两篇一字不差）

| 方法 | 路径 | 入参 | 返回 |
|---|---|---|---|
| GET | `/tasks` | `_page`、`_limit`、`status`、`q` | `{ items, total, hasNext }` |
| GET | `/tasks/{id}` | — | `Task`，不存在 → 404 |
| POST | `/tasks` | `title/status/priority/assignee/desc` | 201 + `Task` |
| PATCH | `/tasks/{id}` | 上述字段的部分 | 200 + `Task` |
| DELETE | `/tasks/{id}` | — | 204 |
| GET | `/healthz` | — | `{ "status": "ok" }` |

> 09 篇的示例挂在 `/api` 前缀下；这里为了和前端契约完全对齐，直接挂根路径 `/tasks`。

```go
// internal/task/model.go
package task

import "time"

type Task struct {
	ID        string    `json:"id"`
	Title     string    `json:"title"`
	Status    string    `json:"status"`
	Priority  string    `json:"priority"`
	Assignee  string    `json:"assignee"`
	Desc      string    `json:"desc"`
	CreatedAt time.Time `json:"createdAt"`
}

type Query struct {
	Page    int
	Limit   int
	Status  string
	Keyword string
}

// 仓储接口定义在业务层（使用方），实现可以在内存、SQLite、Postgres 之间换
type Repo interface {
	List(ctx context.Context, q Query) ([]Task, int, error)
	Get(ctx context.Context, id string) (Task, error)
	Create(ctx context.Context, t Task) (Task, error)
	Update(ctx context.Context, id string, patch map[string]any) (Task, error)
	Delete(ctx context.Context, id string) error
}
```

### 内存实现（阶段 1 够用，阶段 2 会被替换）

```go
// internal/task/store.go
package task

import (
	"context"
	"sort"
	"strings"
	"sync"
	"time"

	"example.com/task-api/internal/httpx"
)

type MemoryStore struct {
	mu    sync.RWMutex
	items map[string]Task
}

func NewMemoryStore() *MemoryStore { return &MemoryStore{items: make(map[string]Task)} }

func (s *MemoryStore) List(ctx context.Context, q Query) ([]Task, int, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()

	matched := make([]Task, 0, len(s.items))
	for _, t := range s.items {
		if q.Status != "" && q.Status != "all" && t.Status != q.Status {
			continue
		}
		if q.Keyword != "" && !strings.Contains(t.Title, q.Keyword) {
			continue
		}
		matched = append(matched, t)
	}
	sort.Slice(matched, func(i, j int) bool { return matched[i].CreatedAt.After(matched[j].CreatedAt) })

	total := len(matched)
	start := (q.Page - 1) * q.Limit
	if start > total {
		start = total
	}
	end := start + q.Limit
	if end > total {
		end = total
	}
	return matched[start:end], total, nil
}

func (s *MemoryStore) Get(ctx context.Context, id string) (Task, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	t, ok := s.items[id]
	if !ok {
		return Task{}, httpx.NotFound("任务不存在")
	}
	return t, nil
}

func (s *MemoryStore) Create(ctx context.Context, t Task) (Task, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t.ID = newID()
	t.CreatedAt = time.Now().UTC()
	s.items[t.ID] = t
	return t, nil
}

func (s *MemoryStore) Update(ctx context.Context, id string, patch map[string]any) (Task, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t, ok := s.items[id]
	if !ok {
		return Task{}, httpx.NotFound("任务不存在")
	}
	if v, ok := patch["title"].(string); ok {
		t.Title = v
	}
	if v, ok := patch["status"].(string); ok {
		t.Status = v
	}
	if v, ok := patch["assignee"].(string); ok {
		t.Assignee = v
	}
	s.items[id] = t
	return t, nil
}

func (s *MemoryStore) Delete(ctx context.Context, id string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.items[id]; !ok {
		return httpx.NotFound("任务不存在")
	}
	delete(s.items, id)
	return nil
}
```

### 校验：入参边界在这一层挡住

```go
// internal/task/handler.go（片段）
type createInput struct {
	Title    string `json:"title"`
	Status   string `json:"status"`
	Priority string `json:"priority"`
	Assignee string `json:"assignee"`
	Desc     string `json:"desc"`
}

func (in *createInput) validate() map[string]string {
	errs := map[string]string{}
	title := strings.TrimSpace(in.Title)
	switch {
	case title == "":
		errs["title"] = "标题不能为空"
	case len([]rune(title)) < 4:
		errs["title"] = "标题至少 4 个字"
	case len([]rune(title)) > 60:
		errs["title"] = "标题不要超过 60 个字"
	}
	if strings.TrimSpace(in.Assignee) == "" {
		errs["assignee"] = "请填写负责人"
	}
	if in.Status == "" {
		in.Status = "todo"
	}
	if in.Status != "todo" && in.Status != "doing" && in.Status != "done" {
		errs["status"] = "状态只能是 todo/doing/done"
	}
	if len(errs) > 0 {
		return errs
	}
	return nil
}
```

> `len([]rune(s))` 才是"字符数"；`len(s)` 是字节数，中文一个字占 3 字节——**这是 Go 里最容易写错的长度判断**。

## 阶段 2：换 SQLite，业务层零改动（2 小时）

### 建表迁移

```sql
-- migrations/001_init.sql
create table if not exists tasks (
	id         text primary key,
	title      text not null,
	status     text not null default 'todo',
	priority   text not null default 'mid',
	assignee   text not null,
	descr      text not null default '',
	created_at timestamp not null,
	updated_at timestamp not null,
	deleted_at timestamp
);
create index if not exists idx_tasks_status_created on tasks(status, created_at desc);
```

> 列名用 `descr` 而不是 `desc`：`desc` 是 SQL 关键字，很多库要加引号，容易踩坑。

### 用同一个 `Repo` 接口实现 SQLite 版

```go
// internal/task/sqlstore.go
package task

type SQLStore struct{ db *sql.DB }

func NewSQLStore(db *sql.DB) *SQLStore { return &SQLStore{db: db} }

func (s *SQLStore) List(ctx context.Context, q Query) ([]Task, int, error) {
	where := "where deleted_at is null"
	args := []any{}
	if q.Status != "" && q.Status != "all" {
		where += " and status = ?"
		args = append(args, q.Status)
	}
	if q.Keyword != "" {
		where += " and title like ?"
		args = append(args, "%"+q.Keyword+"%")
	}

	var total int
	if err := s.db.QueryRowContext(ctx, "select count(*) from tasks "+where, args...).Scan(&total); err != nil {
		return nil, 0, err
	}

	rows, err := s.db.QueryContext(ctx,
		"select id, title, status, priority, assignee, descr, created_at from tasks "+where+
			" order by created_at desc limit ? offset ?",
		append(args, q.Limit, (q.Page-1)*q.Limit)...)
	if err != nil {
		return nil, 0, err
	}
	defer rows.Close()

	var out []Task
	for rows.Next() {
		var t Task
		if err := rows.Scan(&t.ID, &t.Title, &t.Status, &t.Priority, &t.Assignee, &t.Desc, &t.CreatedAt); err != nil {
			return nil, 0, err
		}
		out = append(out, t)
	}
	if err := rows.Err(); err != nil {
		return nil, 0, err
	}
	return out, total, nil
}

func (s *SQLStore) Delete(ctx context.Context, id string) error {
	res, err := s.db.ExecContext(ctx,
		"update tasks set deleted_at = ? where id = ? and deleted_at is null", time.Now().UTC(), id)
	if err != nil {
		return err
	}
	n, _ := res.RowsAffected()
	if n == 0 {
		return httpx.NotFound("任务不存在")
	}
	return nil
}
```

## 常见坑

1. **`len(s)` 当字符数**：中文一个字 3 字节，长度校验会误判。用 `len([]rune(s))` 或 `utf8.RuneCountInString(s)`。
2. **`json.Marshal` 出来的时间是 RFC3339**：前端如果按 `yyyy-MM-dd` 解析会失败，要么前端兼容，要么自定义 `MarshalJSON`。
3. **内存版排序不稳定**：`sort.Slice` 不是稳定排序，同一时间创建的元素顺序会变；需要稳定就用 `sort.SliceStable` 或加 `id` 兜底排序。
4. **SQLite 并发写**：默认单写锁，高并发会 `database is locked`。用 WAL 模式（`?_pragma=journal_mode(WAL)`）或换 Postgres。
5. **`desc` 是 SQL 关键字**：列名改用 `descr`，或每次查询都加引号（更容易忘）。
6. **忘了 `rows.Close()`**：连接池耗尽，症状是"跑一会儿就卡住"。
7. **`Shutdown` 之前没停止后台任务**：定时任务还在跑，进程退不掉。先 `cancel()` 再 `Shutdown`。
8. **`-ldflags="-s -w"` 后无法 pprof 定位**：需要排障的构建版本别加，或用 `-buildvcs` 保留信息。
