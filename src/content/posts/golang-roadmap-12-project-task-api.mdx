> [!NOTE]
> 这是本模块的毕业项目，也是**同一份契约的第三个实现**：
> [React 篇](/blog/posts/react-roadmap-17-project-task-board/) 写前端、[Node 篇](/blog/posts/node-roadmap-13-project-task-api/) 写 NestJS 版、这一篇用 **Go 标准库**写。
>
> 任务资源使用字符串 ID，列表返回分页对象，删除成功返回 204。React API 层兼容 json-server 数组和正式分页响应，TypeScript SDK 使用正式分页契约。Go 示例先在本机验证，Node 的认证版本需要额外携带登录令牌。

---

## 做到什么算过

1. 能用 `net/http` + `ServeMux` 写出 `GET /tasks/{id}` 级别的路由，不需要框架。
2. 分页、筛选、搜索的入参解析有边界保护（负数、超上限、非数字都不会 500）。
3. 错误统一：404 / 422 / 500 三种情形返回同一结构的 JSON，且带 `requestId`。
4. 数据层从内存换到 SQLite，**业务层代码不用改**（靠接口隔离）。
5. 内存与 SQLite 共用一份 HTTP 契约测试，包含字段错误、404、分页与删除后的查询。
6. 用 `CGO_ENABLED=0 go build` 出一个静态二进制，扔到服务器能直接跑。
7. 优雅关闭时停止接收新连接，等待在途请求完成，超过 10 秒再强制关闭。
8. 前端（React 那篇）把 base 指过来，列表/新建/删除全部可用，**前端零改动**。

## 共享的任务资源契约

| 方法 | 路径 | 入参 | 返回 |
|---|---|---|---|
| GET | `/tasks` | `_page`、`_limit`、`status`、`q` | `{ items, total, hasNext }` |
| GET | `/tasks/{id}` | — | `Task`，不存在 → 404 |
| POST | `/tasks` | `title/status/priority/assignee/desc` | 201 + `Task` |
| PATCH | `/tasks/{id}` | 上述字段的部分 | 200 + `Task` |
| DELETE | `/tasks/{id}` | — | 204 |
| GET | `/healthz` | — | `{ "status": "ok" }` |

> 09 篇的示例挂在 `/api` 前缀下；这里为了和前端契约完全对齐，直接挂根路径 `/tasks`。

### 模型、仓储边界与字段校验

创建时补 todo、mid 和服务端 UTC 时间。PATCH 只修改传入字段，在仓储的锁或事务内读取当前数据、合并和校验，因此同时修改不同字段时能保留两次变更。标题长度按 Unicode 码点计算，组合字形可能由多个码点构成。

```go title="internal/task/model.go"
package task

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"strings"
	"time"
	"unicode/utf8"
)

type Task struct {
	ID        string    `json:"id"`
	Title     string    `json:"title"`
	Status    string    `json:"status"`
	Priority  string    `json:"priority"`
	Assignee  string    `json:"assignee"`
	Desc      string    `json:"desc"`
	CreatedAt time.Time `json:"createdAt"`
}

type Query struct {
	Page, Limit     int
	Status, Keyword string
}
type Page struct {
	Items   []Task `json:"items"`
	Total   int    `json:"total"`
	HasNext bool   `json:"hasNext"`
}
type APIError struct {
	Status  int
	Message string
	Fields  map[string][]string
}

func (e *APIError) Error() string { return e.Message }
func missing() error              { return &APIError{Status: 404, Message: "任务不存在"} }

type Repo interface {
	List(context.Context, Query) (Page, error)
	Get(context.Context, string) (Task, error)
	Create(context.Context, map[string]string) (Task, error)
	Update(context.Context, string, map[string]string) (Task, error)
	Delete(context.Context, string) error
}

func NewID() string {
	var id [16]byte
	if _, err := rand.Read(id[:]); err != nil {
		panic(err)
	}
	return hex.EncodeToString(id[:])
}

func apply(t Task, fields map[string]string) (Task, error) {
	for key, value := range fields {
		switch key {
		case "title":
			t.Title = strings.TrimSpace(value)
		case "status":
			t.Status = value
		case "priority":
			t.Priority = value
		case "assignee":
			t.Assignee = strings.TrimSpace(value)
		case "desc":
			t.Desc = value
		}
	}
	errors := map[string][]string{}
	if n := utf8.RuneCountInString(t.Title); n < 4 || n > 60 {
		errors["title"] = []string{"标题应为 4 到 60 个 Unicode 码点"}
	}
	if t.Assignee == "" {
		errors["assignee"] = []string{"请填写负责人"}
	}
	if utf8.RuneCountInString(t.Desc) > 500 {
		errors["desc"] = []string{"描述最多 500 个 Unicode 码点"}
	}
	if t.Status != "todo" && t.Status != "doing" && t.Status != "done" {
		errors["status"] = []string{"状态只能是 todo/doing/done"}
	}
	if t.Priority != "low" && t.Priority != "mid" && t.Priority != "high" {
		errors["priority"] = []string{"优先级只能是 low/mid/high"}
	}
	if len(errors) > 0 {
		return Task{}, &APIError{422, "提交的数据不合法", errors}
	}
	return t, nil
}

func create(fields map[string]string) (Task, error) {
	return apply(Task{ID: NewID(), Status: "todo", Priority: "mid", CreatedAt: time.Now().UTC()}, fields)
}
```

### 内存仓储

列表按创建时间和 ID 排序，空列表序列化为 `[]`。写入和更新在锁内完成，读取返回值对象。

```go title="internal/task/memory.go"
package task

import (
	"context"
	"sort"
	"strings"
	"sync"
)

type MemoryStore struct {
	mu    sync.RWMutex
	items map[string]Task
}

func NewMemoryStore() *MemoryStore { return &MemoryStore{items: map[string]Task{}} }

func (s *MemoryStore) List(_ context.Context, q Query) (Page, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	items := []Task{}
	for _, t := range s.items {
		if (q.Status == "" || t.Status == q.Status) && strings.Contains(t.Title, q.Keyword) {
			items = append(items, t)
		}
	}
	sort.Slice(items, func(i, j int) bool {
		if items[i].CreatedAt.Equal(items[j].CreatedAt) {
			return items[i].ID > items[j].ID
		}
		return items[i].CreatedAt.After(items[j].CreatedAt)
	})
	total := len(items)
	start := min((q.Page-1)*q.Limit, total)
	end := min(start+q.Limit, total)
	return Page{items[start:end], total, end < total}, nil
}
func (s *MemoryStore) Get(_ context.Context, id string) (Task, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	t, ok := s.items[id]
	if !ok {
		return Task{}, missing()
	}
	return t, nil
}
func (s *MemoryStore) Create(_ context.Context, fields map[string]string) (Task, error) {
	t, err := create(fields)
	if err != nil {
		return Task{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.items[t.ID] = t
	return t, nil
}
func (s *MemoryStore) Update(_ context.Context, id string, fields map[string]string) (Task, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t, ok := s.items[id]
	if !ok {
		return Task{}, missing()
	}
	t, err := apply(t, fields)
	if err != nil {
		return Task{}, err
	}
	s.items[id] = t
	return t, nil
}
func (s *MemoryStore) Delete(_ context.Context, id string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.items[id]; !ok {
		return missing()
	}
	delete(s.items, id)
	return nil
}
```

### SQLite 仓储

启动时建表，单连接串行写事务，列表用同一读事务读取 total 和 items。生产环境的结构变更应改用版本化迁移。这里的单连接配置适用于本地单进程实验，跨进程写入仍需 busy timeout、重试与容量验证。筛选统一使用区分大小写的子串匹配，SQL 通过参数绑定传值。

```go title="internal/task/sqlite.go"
package task

import (
	"context"
	"database/sql"
	"errors"
	_ "modernc.org/sqlite"
	"time"
)

type SQLStore struct{ db *sql.DB }

func OpenSQLite(ctx context.Context, dsn string) (*sql.DB, error) {
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	// 本实验单进程单连接，将写事务串行化。跨进程写入仍需 busy timeout 等策略。
	db.SetMaxOpenConns(1)
	db.SetMaxIdleConns(1)
	if err = db.PingContext(ctx); err == nil {
		_, err = db.ExecContext(ctx, `CREATE TABLE IF NOT EXISTS tasks (
id TEXT PRIMARY KEY, title TEXT NOT NULL, status TEXT NOT NULL,
priority TEXT NOT NULL, assignee TEXT NOT NULL, descr TEXT NOT NULL,
created_at TEXT NOT NULL, deleted_at TEXT);
CREATE INDEX IF NOT EXISTS idx_tasks_status_created ON tasks(status,created_at DESC,id DESC);`)
	}
	if err != nil {
		db.Close()
		return nil, err
	}
	return db, nil
}
func NewSQLStore(db *sql.DB) *SQLStore { return &SQLStore{db} }

const columns = "id,title,status,priority,assignee,descr,created_at"
const storedTime = "2006-01-02T15:04:05.000000000Z"

type scanner interface{ Scan(...any) error }

func scanTask(row scanner) (Task, error) {
	var t Task
	var created string
	err := row.Scan(&t.ID, &t.Title, &t.Status, &t.Priority, &t.Assignee, &t.Desc, &created)
	if errors.Is(err, sql.ErrNoRows) {
		return t, missing()
	}
	if err != nil {
		return t, err
	}
	t.CreatedAt, err = time.Parse(time.RFC3339Nano, created)
	return t, err
}
func (s *SQLStore) Get(ctx context.Context, id string) (Task, error) {
	return scanTask(s.db.QueryRowContext(ctx, "SELECT "+columns+" FROM tasks WHERE id=? AND deleted_at IS NULL", id))
}
func (s *SQLStore) List(ctx context.Context, q Query) (Page, error) {
	tx, err := s.db.BeginTx(ctx, &sql.TxOptions{ReadOnly: true})
	if err != nil {
		return Page{}, err
	}
	defer tx.Rollback()
	where := " WHERE deleted_at IS NULL AND (?='' OR status=?) AND instr(title,?)>0"
	args := []any{q.Status, q.Status, q.Keyword}
	page := Page{Items: []Task{}}
	if err := tx.QueryRowContext(ctx, "SELECT count(*) FROM tasks"+where, args...).Scan(&page.Total); err != nil {
		return Page{}, err
	}
	args = append(args, q.Limit, (q.Page-1)*q.Limit)
	rows, err := tx.QueryContext(ctx, "SELECT "+columns+" FROM tasks"+where+" ORDER BY created_at DESC,id DESC LIMIT ? OFFSET ?", args...)
	if err != nil {
		return Page{}, err
	}
	defer rows.Close()
	for rows.Next() {
		t, err := scanTask(rows)
		if err != nil {
			return Page{}, err
		}
		page.Items = append(page.Items, t)
	}
	if err := rows.Err(); err != nil {
		return Page{}, err
	}
	if err := rows.Close(); err != nil {
		return Page{}, err
	}
	page.HasNext = (q.Page-1)*q.Limit+len(page.Items) < page.Total
	return page, tx.Commit()
}
func (s *SQLStore) Create(ctx context.Context, fields map[string]string) (Task, error) {
	t, err := create(fields)
	if err != nil {
		return Task{}, err
	}
	_, err = s.db.ExecContext(ctx, "INSERT INTO tasks("+columns+") VALUES(?,?,?,?,?,?,?)", t.ID, t.Title, t.Status, t.Priority, t.Assignee, t.Desc, t.CreatedAt.Format(storedTime))
	return t, err
}
func (s *SQLStore) Update(ctx context.Context, id string, fields map[string]string) (Task, error) {
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return Task{}, err
	}
	defer tx.Rollback()
	t, err := scanTask(tx.QueryRowContext(ctx, "SELECT "+columns+" FROM tasks WHERE id=? AND deleted_at IS NULL", id))
	if err != nil {
		return Task{}, err
	}
	t, err = apply(t, fields)
	if err != nil {
		return Task{}, err
	}
	_, err = tx.ExecContext(ctx, "UPDATE tasks SET title=?,status=?,priority=?,assignee=?,descr=? WHERE id=? AND deleted_at IS NULL", t.Title, t.Status, t.Priority, t.Assignee, t.Desc, id)
	if err != nil {
		return Task{}, err
	}
	return t, tx.Commit()
}
func (s *SQLStore) Delete(ctx context.Context, id string) error {
	result, err := s.db.ExecContext(ctx, "UPDATE tasks SET deleted_at=? WHERE id=? AND deleted_at IS NULL", time.Now().UTC().Format(time.RFC3339Nano), id)
	if err != nil {
		return err
	}
	count, err := result.RowsAffected()
	if err != nil {
		return err
	}
	if count == 0 {
		return missing()
	}
	return nil
}
```

### HTTP 与统一错误

分页参数先解析再限制范围，最大页码为一百万，页大小最多一百。非法 JSON 返回 400，字段校验返回 422，找不到任务返回 404。PATCH 的 null 或非字符串值返回 400。CORS 允许配置的单个前端 origin，浏览器预检也能通过。

```go title="internal/task/handler.go"
package task

import (
	"encoding/json"
	"errors"
	"io"
	"log/slog"
	"mime"
	"net/http"
	"strconv"
	"strings"
)

func ParsePaging(r *http.Request) Query {
	page, err := strconv.Atoi(r.URL.Query().Get("_page"))
	if err != nil || page < 1 || page > 1000000 {
		page = 1
	}
	limit, err := strconv.Atoi(r.URL.Query().Get("_limit"))
	if err != nil || limit < 1 {
		limit = 20
	}
	limit = min(limit, 100)
	status := r.URL.Query().Get("status")
	if status == "all" {
		status = ""
	}
	return Query{page, limit, status, r.URL.Query().Get("q")}
}
func writeJSON(w http.ResponseWriter, status int, value any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	if err := json.NewEncoder(w).Encode(value); err != nil {
		slog.Error("write response", "err", err)
	}
}
func writeError(w http.ResponseWriter, err error) {
	e := &APIError{Status: 500, Message: "服务器内部错误"}
	if !errors.As(err, &e) {
		slog.Error("request failed", "err", err)
		e = &APIError{Status: 500, Message: "服务器内部错误"}
	}
	writeJSON(w, e.Status, struct {
		Status    int                 `json:"statusCode"`
		Message   string              `json:"message"`
		Errors    map[string][]string `json:"errors,omitempty"`
		RequestID string              `json:"requestId"`
	}{e.Status, e.Message, e.Fields, w.Header().Get("X-Request-Id")})
}
func decodeInput(w http.ResponseWriter, r *http.Request) (map[string]string, error) {
	contentType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
	if err != nil || contentType != "application/json" {
		return nil, &APIError{Status: 415, Message: "需要 application/json"}
	}
	r.Body = http.MaxBytesReader(w, r.Body, 65536)
	decoder := json.NewDecoder(r.Body)
	var fields map[string]json.RawMessage
	if err := decoder.Decode(&fields); err != nil || fields == nil {
		return nil, &APIError{Status: 400, Message: "请求 JSON 不合法"}
	}
	if err := decoder.Decode(new(any)); err != io.EOF {
		return nil, &APIError{Status: 400, Message: "请求只能包含一个 JSON 对象"}
	}
	values := map[string]string{}
	for key, raw := range fields {
		switch key {
		case "title", "status", "priority", "assignee", "desc", "createdAt":
		default:
			return nil, &APIError{Status: 400, Message: "未知字段 " + key}
		}
		if key == "createdAt" {
			continue
		} // 接受旧 mock 的创建时间字段，最终时间由服务端生成。
		var value string
		if string(raw) == "null" || json.Unmarshal(raw, &value) != nil {
			return nil, &APIError{Status: 400, Message: key + " 必须是字符串"}
		}
		values[key] = value
	}
	return values, nil
}

func NewHandler(repo Repo, origin string) http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, r *http.Request) { writeJSON(w, 200, map[string]string{"status": "ok"}) })
	mux.HandleFunc("GET /tasks", func(w http.ResponseWriter, r *http.Request) {
		page, err := repo.List(r.Context(), ParsePaging(r))
		if err != nil {
			writeError(w, err)
			return
		}
		writeJSON(w, 200, page)
	})
	mux.HandleFunc("GET /tasks/{id}", func(w http.ResponseWriter, r *http.Request) {
		t, err := repo.Get(r.Context(), r.PathValue("id"))
		if err != nil {
			writeError(w, err)
			return
		}
		writeJSON(w, 200, t)
	})
	save := func(w http.ResponseWriter, r *http.Request) {
		fields, err := decodeInput(w, r)
		if err != nil {
			writeError(w, err)
			return
		}
		var t Task
		status := http.StatusOK
		if r.Method == "POST" {
			t, err = repo.Create(r.Context(), fields)
			status = http.StatusCreated
		} else {
			t, err = repo.Update(r.Context(), r.PathValue("id"), fields)
		}
		if err != nil {
			writeError(w, err)
			return
		}
		writeJSON(w, status, t)
	}
	mux.HandleFunc("POST /tasks", save)
	mux.HandleFunc("PATCH /tasks/{id}", save)
	mux.HandleFunc("DELETE /tasks/{id}", func(w http.ResponseWriter, r *http.Request) {
		if err := repo.Delete(r.Context(), r.PathValue("id")); err != nil {
			writeError(w, err)
			return
		}
		w.WriteHeader(http.StatusNoContent)
	})
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("X-Request-Id", NewID())
		defer func() {
			if err := recover(); err != nil {
				slog.Error("panic", "err", err)
				writeError(w, errors.New("panic"))
			}
		}()
		if r.Header.Get("Origin") == origin && origin != "" {
			w.Header().Set("Access-Control-Allow-Origin", origin)
			w.Header().Set("Vary", "Origin")
			w.Header().Set("Access-Control-Allow-Methods", "GET,POST,PATCH,DELETE,OPTIONS")
			w.Header().Set("Access-Control-Allow-Headers", "Content-Type,Authorization")
			w.Header().Set("Access-Control-Expose-Headers", "X-Request-Id")
		}
		if r.Method == "OPTIONS" {
			w.WriteHeader(204)
			return
		}
		slog.Info("request", "requestId", w.Header().Get("X-Request-Id"), "method", r.Method, "path", r.URL.Path)
		// Go 1.22 的 ServeMux 原生处理方法路由和路径参数。
		if strings.HasSuffix(r.URL.Path, "/") && r.URL.Path != "/" {
			r.URL.Path = strings.TrimSuffix(r.URL.Path, "/")
		}
		mux.ServeHTTP(w, r)
	})
}
```

### 装配与优雅关闭

未设置 DATABASE_URL 时使用内存，有值时使用 SQLite。HTTP 处理层接收同一个 Repo 接口。收到退出信号后最多等待 10 秒，然后关闭连接，数据库连接在 HTTP 服务退出后释放。

```go title="cmd/api/main.go"
package main

import (
	"context"
	"errors"
	"example.com/task-api/internal/task"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"
)

func main() {
	slog.SetDefault(slog.New(slog.NewJSONHandler(os.Stdout, nil)))
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()
	var repo task.Repo = task.NewMemoryStore()
	if dsn := os.Getenv("DATABASE_URL"); dsn != "" {
		db, err := task.OpenSQLite(ctx, dsn)
		if err != nil {
			slog.Error("database", "err", err)
			os.Exit(1)
		}
		defer db.Close()
		repo = task.NewSQLStore(db)
	}
	addr := os.Getenv("APP_ADDR")
	if addr == "" {
		addr = "127.0.0.1:8080"
	}
	origin := os.Getenv("CORS_ORIGIN")
	if origin == "" {
		origin = "http://localhost:5173"
	}
	server := &http.Server{Addr: addr, Handler: task.NewHandler(repo, origin), ReadHeaderTimeout: 5 * time.Second, ReadTimeout: 15 * time.Second, WriteTimeout: 30 * time.Second, IdleTimeout: 60 * time.Second}
	failed := make(chan error, 1)
	go func() { failed <- server.ListenAndServe() }()
	slog.Info("listening", "addr", addr)
	select {
	case err := <-failed:
		if !errors.Is(err, http.ErrServerClosed) {
			slog.Error("server", "err", err)
			os.Exit(1)
		}
	case <-ctx.Done():
		slog.Info("shutting down")
		shutdown, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		if err := server.Shutdown(shutdown); err != nil {
			slog.Error("shutdown", "err", err)
			server.Close()
		}
	}
}
```

## 常见坑

1. **`len(s)` 当字符数**：中文一个字 3 字节，长度校验会误判。用 `len([]rune(s))` 或 `utf8.RuneCountInString(s)`。
2. **`json.Marshal` 出来的时间是 RFC3339**：前端如果按 `yyyy-MM-dd` 解析会失败，要么前端兼容，要么自定义 `MarshalJSON`。
3. **分页排序不确定**：同一时间创建的任务使用 ID 兜底排序。Map 遍历本身无序，单独换稳定排序算法无法固定输入顺序。
4. **SQLite 并发写**：默认单写锁，高并发会 `database is locked`。用 WAL 模式（`?_pragma=journal_mode(WAL)`）或换 Postgres。
5. **`desc` 是 SQL 关键字**：列名改用 `descr`，或每次查询都加引号（更容易忘）。
6. **忘了 `rows.Close()`**：连接池耗尽，症状是"跑一会儿就卡住"。
7. **`Shutdown` 之前没停止后台任务**：定时任务还在跑，进程退不掉。先 `cancel()` 再 `Shutdown`。
8. **构建信息不完整**：保留与发布版本匹配的二进制、源码 commit 和构建参数，以便分析 profile。DWARF 调试信息与 pprof 的 Go 符号信息用途不同。
