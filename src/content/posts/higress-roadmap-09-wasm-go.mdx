```go
package main

  "github.com/higress-group/proxy-wasm-go-sdk/proxywasm/types"

type HelloConfig struct { Message string }

func main() {
  wrapper.SetCtx("hello-plugin", wrapper.ParseConfig(parseConfig), wrapper.ProcessRequestHeaders(onHeaders))
}
func parseConfig(raw gjson.Result, config *HelloConfig) error {
  config.Message = raw.Get("message").String()
  if config.Message == "" { config.Message = "Hello from Higress Wasm Plugin!" }

func onHeaders(_ wrapper.HttpContext, config HelloConfig) types.Action {
  if err := proxywasm.ReplaceHttpRequestHeader("x-hello-message", config.Message); err != nil {
    proxywasm.LogError(err.Error())
  }
  return types.ActionContinue
}
```

## SDK 核心 API

### wrapper 与 proxywasm 的分工

wasm-go 1.2.1 用 `wrapper.ParseConfig` 注册配置解析，用 `wrapper.ProcessRequestHeaders` 注册请求头钩子。配置函数返回 error，请求函数返回 `types.Action`。Header 操作和异步调用使用 `proxywasm` 的宿主 API，暂停请求返回 `types.ActionPause`，回调成功后调用 `proxywasm.ResumeHttpRequest()`。

下面的 Hello、JWT 与外部 HTTP 示例分别放进独立模块，分别编译各自的 main。JWT 示例已经包含完整的依赖、测试和状态处理。 [wasm-go 1.2.1 接口源码](https://github.com/higress-group/wasm-go/blob/v1.2.1/pkg/wrapper/plugin_wrapper.go)

### proxywasm 底层 API

```go
import "github.com/higress-group/proxy-wasm-go-sdk/proxywasm"

// 日志
proxywasm.LogDebug("debug message")
proxywasm.LogInfo("info message")
proxywasm.LogWarn("warning message")
proxywasm.LogError("error message")

// Header 操作
proxywasm.GetHttpRequestHeader("key")
proxywasm.AddHttpRequestHeader("key", "value")
proxywasm.ReplaceHttpRequestHeader("key", "value")
proxywasm.RemoveHttpRequestHeader("key")

// HTTP 调用（异步）
proxywasm.DispatchHttpCall(
	cluster string,           // 上游集群名
	headers [][2]string,      // 请求头
	body []byte,              // 请求体
	trailers [][2]string,     // Trailer
	timeout uint32,           // 超时（毫秒）
	callback func(numHeaders, bodySize, numTrailers int),  // 回调
)

// 发送响应
proxywasm.SendHttpResponse(
	statusCode uint32,
	headers [][2]string,
	body []byte,
	grpcStatus int32,
)
```

### 公钥解析与验签

JWKS 按 `kid` 保存 RSA 公钥，限定 RS256、issuer、audience、到期时间与 subject。解析 header 只是为了选 key，最终身份必须通过签名与 claims 校验。

```go title="jwks.go"
package main

import (
	"crypto/rsa"
	"encoding/base64"
	"encoding/json"
	"errors"
	"github.com/golang-jwt/jwt/v5"
	"math/big"
	"strings"
)

type jwk struct {
	Kty string `json:"kty"`
	Kid string `json:"kid"`
	Alg string `json:"alg"`
	Use string `json:"use"`
	N   string `json:"n"`
	E   string `json:"e"`
}

func decodeJWKS(body []byte) (map[string]*rsa.PublicKey, error) {
	var set struct {
		Keys []jwk `json:"keys"`
	}
	if err := json.Unmarshal(body, &set); err != nil {
		return nil, err
	}
	if len(set.Keys) == 0 || len(set.Keys) > 100 {
		return nil, errors.New("invalid key count")
	}
	keys := map[string]*rsa.PublicKey{}
	for _, key := range set.Keys {
		if key.Kty != "RSA" || (key.Alg != "" && key.Alg != "RS256") || (key.Use != "" && key.Use != "sig") {
			continue
		}
		n, err := base64.RawURLEncoding.DecodeString(key.N)
		if err != nil {
			return nil, err
		}
		e, err := base64.RawURLEncoding.DecodeString(key.E)
		if err != nil || len(e) == 0 || len(e) > 4 {
			return nil, errors.New("invalid exponent")
		}
		exponent := new(big.Int).SetBytes(e).Int64()
		modulus := new(big.Int).SetBytes(n)
		if key.Kid == "" || modulus.BitLen() < 2048 || modulus.BitLen() > 8192 || exponent < 3 || exponent > 2147483647 || exponent%2 == 0 {
			return nil, errors.New("invalid RSA key")
		}
		if _, ok := keys[key.Kid]; ok {
			return nil, errors.New("duplicate kid")
		}
		keys[key.Kid] = &rsa.PublicKey{N: modulus, E: int(exponent)}
	}
	if len(keys) == 0 {
		return nil, errors.New("no RS256 signing key")
	}
	return keys, nil
}

func tokenKid(raw string) (string, error) {
	if len(raw) > 16384 {
		return "", errors.New("token too large")
	}
	parts := strings.Split(raw, ".")
	if len(parts) != 3 {
		return "", errors.New("invalid token")
	}
	header, err := base64.RawURLEncoding.DecodeString(parts[0])
	if err != nil {
		return "", err
	}
	var fields struct {
		Alg string `json:"alg"`
		Kid string `json:"kid"`
	}
	if err := json.Unmarshal(header, &fields); err != nil {
		return "", err
	}
	if fields.Alg != "RS256" || fields.Kid == "" {
		return "", errors.New("RS256 and kid required")
	}
	return fields.Kid, nil
}

func verifyToken(raw string, key *rsa.PublicKey, issuer, audience string) (string, error) {
	claims := &jwt.RegisteredClaims{}
	_, err := jwt.ParseWithClaims(raw, claims, func(*jwt.Token) (any, error) { return key, nil },
		jwt.WithValidMethods([]string{"RS256"}), jwt.WithIssuer(issuer),
		jwt.WithAudience(audience), jwt.WithExpirationRequired())
	if err != nil {
		return "", err
	}
	if claims.Subject == "" || strings.ContainsAny(claims.Subject, "\r\n") {
		return "", errors.New("invalid subject")
	}
	return claims.Subject, nil
}
```

### 请求暂停、异步加载与缓存

每个 worker 的配置持有独立缓存。有效 key 命中直接验签，过期或未知 kid 发起一次异步加载，请求暂停到回调完成。刷新间隔下限 5 秒，成功后缓存 5 分钟。刷新失败返回 503，验签失败返回 401。并发冷启动期间可能有请求得到 503，调用方按退避策略重试。

```go title="main.go"
package main

import (
	"crypto/rsa"
	"errors"
	"github.com/higress-group/proxy-wasm-go-sdk/proxywasm"
	"github.com/higress-group/proxy-wasm-go-sdk/proxywasm/types"
	"github.com/higress-group/wasm-go/pkg/wrapper"
	"github.com/tidwall/gjson"
	"strings"
	"time"
)

type keyCache struct {
	keys       map[string]*rsa.PublicKey
	expires    time.Time
	retryAfter time.Time
}
type JWTConfig struct {
	issuer, audience, cluster, host, path string
	cache                                 *keyCache
}

func main() {
	wrapper.SetCtx("jwt-auth", wrapper.ParseConfig(parseConfig), wrapper.ProcessRequestHeaders(onHeaders))
}
func parseConfig(raw gjson.Result, config *JWTConfig) error {
	config.issuer = raw.Get("issuer").String()
	config.audience = raw.Get("audience").String()
	config.cluster = raw.Get("jwks_cluster").String()
	config.host = raw.Get("jwks_host").String()
	config.path = raw.Get("jwks_path").String()
	if config.issuer == "" || config.audience == "" || config.cluster == "" || config.host == "" || !strings.HasPrefix(config.path, "/") {
		return errors.New("issuer/audience/jwks_cluster/jwks_host/jwks_path required")
	}
	config.cache = &keyCache{keys: map[string]*rsa.PublicKey{}}
	return nil
}
func reject(status uint32) types.Action {
	_ = proxywasm.SendHttpResponse(status, [][2]string{{"content-type", "application/json"}}, []byte(`{"message":"authentication unavailable or rejected"}`), -1)
	return types.ActionPause
}
func authorize(raw, kid string, config JWTConfig) bool {
	key := config.cache.keys[kid]
	if key == nil {
		reject(401)
		return false
	}
	sub, err := verifyToken(raw, key, config.issuer, config.audience)
	if err != nil {
		reject(401)
		return false
	}
	if err := proxywasm.ReplaceHttpRequestHeader("x-authenticated-sub", sub); err != nil {
		reject(503)
		return false
	}
	return true
}
func onHeaders(_ wrapper.HttpContext, config JWTConfig) types.Action {
	// 先去掉客户端伪造的身份，只允许本插件写入这个上游身份头。
	if err := proxywasm.RemoveHttpRequestHeader("x-authenticated-sub"); err != nil && err != types.ErrorStatusNotFound {
		return reject(503)
	}
	header, _ := proxywasm.GetHttpRequestHeader("authorization")
	parts := strings.Fields(header)
	if len(parts) != 2 || !strings.EqualFold(parts[0], "Bearer") {
		return reject(401)
	}
	raw := parts[1]
	kid, err := tokenKid(raw)
	if err != nil {
		return reject(401)
	}
	now := time.Now()
	cache := config.cache
	if cache.keys[kid] != nil && now.Before(cache.expires) {
		if authorize(raw, kid, config) {
			return types.ActionContinue
		}
		return types.ActionPause
	}
	// 每个 worker 至多每 5 秒主动刷新一次，限制未知 kid 的回源频率。
	if now.Before(cache.retryAfter) {
		return reject(503)
	}
	cache.retryAfter = now.Add(5 * time.Second)
	headers := [][2]string{{":method", "GET"}, {":path", config.path}, {":authority", config.host}, {":scheme", "https"}}
	_, err = proxywasm.DispatchHttpCall(config.cluster, headers, nil, nil, 3000, func(_ int, bodySize int, _ int) {
		responseHeaders, headerErr := proxywasm.GetHttpCallResponseHeaders()
		status := ""
		for _, header := range responseHeaders {
			if header[0] == ":status" {
				status = header[1]
			}
		}
		if headerErr != nil || status != "200" || bodySize > 1024*1024 {
			reject(503)
			return
		}
		body, err := proxywasm.GetHttpCallResponseBody(0, bodySize)
		if err != nil {
			reject(503)
			return
		}
		keys, err := decodeJWKS(body)
		if err != nil {
			reject(503)
			return
		}
		// 完整替换缓存，失效或被移除的旧公钥会一起淘汰。
		cache.keys = keys
		cache.expires = time.Now().Add(5 * time.Minute)
		if authorize(raw, kid, config) {
			_ = proxywasm.ResumeHttpRequest()
		}
	})
	if err != nil {
		return reject(503)
	}
	return types.ActionPause
}
```

### 配置与轮换验证

先在 Higress 服务来源里登记 JWKS 的 HTTPS 服务，确认网关能连通并校验证书，再从 Envoy 配置中取得该服务的实际 cluster 名。把 `REPLACE_WITH_JWKS_CLUSTER` 换成这个值。`jwks_host` 是固定认证服务主机，`jwks_path` 来自服务端配置。 [官方 HTTP 调用说明](https://higress.cn/docs/latest/plugins/wasm-dev/wasm17/)

```yaml
apiVersion: extensions.higress.io/v1alpha1
kind: WasmPlugin
metadata:
  name: jwt-auth
  namespace: higress-system
spec:
  url: oci://registry.example.com/plugins/jwt-auth:v1
  phase: AUTHN
  priority: 1000
  defaultConfig:
    issuer: "https://auth.example.com"
    audience: "task-api"
    jwks_cluster: "REPLACE_WITH_JWKS_CLUSTER"
    jwks_host: "auth.example.com"
    jwks_path: "/.well-known/jwks.json"
```

部署时将插件作用范围绑定到要保护的业务路由，公开健康检查使用独立路由。上游只信任网关写入的 `X-Authenticated-Sub`，并通过网络限制外部绕过网关直连业务服务。公钥轮换时，新 JWKS 同时保留新旧 kid，等旧令牌到期与缓存窗口过去再移除旧 key。

### 自动测试与网关验收

```go title="jwks_test.go"
package main

import (
	"crypto/rand"
	"crypto/rsa"
	"encoding/base64"
	"encoding/json"
	"github.com/golang-jwt/jwt/v5"
	"math/big"
	"testing"
	"time"
)

func TestJWKSAndToken(t *testing.T) {
	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		t.Fatal(err)
	}
	set := struct {
		Keys []jwk `json:"keys"`
	}{[]jwk{{Kty: "RSA", Kid: "lab-1", Alg: "RS256", Use: "sig", N: base64.RawURLEncoding.EncodeToString(key.N.Bytes()), E: base64.RawURLEncoding.EncodeToString(big.NewInt(int64(key.E)).Bytes())}}}
	body, _ := json.Marshal(set)
	keys, err := decodeJWKS(body)
	if err != nil {
		t.Fatal(err)
	}
	issue := func(expires time.Time) string {
		token := jwt.NewWithClaims(jwt.SigningMethodRS256, jwt.RegisteredClaims{Issuer: "https://auth.example.com", Subject: "alice", Audience: jwt.ClaimStrings{"task-api"}, ExpiresAt: jwt.NewNumericDate(expires)})
		token.Header["kid"] = "lab-1"
		raw, err := token.SignedString(key)
		if err != nil {
			t.Fatal(err)
		}
		return raw
	}
	valid := issue(time.Now().Add(time.Minute))
	kid, err := tokenKid(valid)
	if err != nil || kid != "lab-1" {
		t.Fatalf("kid=%s err=%v", kid, err)
	}
	if sub, err := verifyToken(valid, keys[kid], "https://auth.example.com", "task-api"); err != nil || sub != "alice" {
		t.Fatalf("sub=%s err=%v", sub, err)
	}
	if _, err := verifyToken(valid, keys[kid], "wrong", "task-api"); err == nil {
		t.Fatal("issuer")
	}
	if _, err := verifyToken(valid, keys[kid], "https://auth.example.com", "wrong"); err == nil {
		t.Fatal("audience")
	}
	if _, err := verifyToken(issue(time.Now().Add(-time.Minute)), keys[kid], "https://auth.example.com", "task-api"); err == nil {
		t.Fatal("expiration")
	}
	if _, err := decodeJWKS([]byte(`{"keys":[]}`)); err == nil {
		t.Fatal("empty keys")
	}
	if _, err := tokenKid("broken"); err == nil {
		t.Fatal("malformed")
	}
}
```

## 实战：HTTP 调用外部服务

### 异步 HTTP 调用

```go
package main

import (
  "encoding/json"
  "errors"
  "github.com/higress-group/wasm-go/pkg/wrapper"
  "github.com/higress-group/proxy-wasm-go-sdk/proxywasm"
  "github.com/higress-group/proxy-wasm-go-sdk/proxywasm/types"
  "github.com/tidwall/gjson"
)

type ExternalConfig struct { Cluster, Host string }
func main() {
  wrapper.SetCtx("external-user", wrapper.ParseConfig(parseConfig), wrapper.ProcessRequestHeaders(onHeaders))
}
func parseConfig(raw gjson.Result, config *ExternalConfig) error {
  config.Cluster = raw.Get("cluster_name").String()
  config.Host = raw.Get("host").String()
  if config.Cluster == "" || config.Host == "" { return errors.New("cluster_name and host required") }
  return nil
}
func fail() types.Action {
  _ = proxywasm.SendHttpResponse(503, nil, []byte("external service unavailable"), -1)
  return types.ActionPause
}
func onHeaders(_ wrapper.HttpContext, config ExternalConfig) types.Action {
  _ = proxywasm.RemoveHttpRequestHeader("x-user-id")
  requestID, _ := proxywasm.GetHttpRequestHeader("x-request-id")
  headers := [][2]string{{":method", "GET"}, {":path", "/api/user-info"}, {":authority", config.Host}, {"x-request-id", requestID}}
  _, err := proxywasm.DispatchHttpCall(config.Cluster, headers, nil, nil, 3000, func(_ int, size int, _ int) {
    headers, err := proxywasm.GetHttpCallResponseHeaders()
    if err != nil || size > 65536 { fail(); return }
    status := ""
    for _, header := range headers { if header[0] == ":status" { status = header[1] } }
    if status != "200" { fail(); return }
    body, err := proxywasm.GetHttpCallResponseBody(0, size)
    var result struct { UserID string `json:"user_id"` }
    if err != nil || json.Unmarshal(body, &result) != nil || result.UserID == "" { fail(); return }
    if err := proxywasm.ReplaceHttpRequestHeader("x-user-id", result.UserID); err != nil { fail(); return }
    _ = proxywasm.ResumeHttpRequest()
  })
  if err != nil { return fail() }
  return types.ActionPause
}
```

```go
log.Warnf("Warning: %v", err)       // 警告
log.Errorf("Error: %v", err)        // 错误

// 查看日志
kubectl logs -n higress-system deploy/higress-gateway -f | grep hello-plugin
```

### CI/CD 自动发布

```yaml
# .github/workflows/release.yml
name: Release Wasm Plugin

on:
  push:
    tags: ['v*']

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - uses: actions/setup-go@v5
        with:
          go-version: '1.24'
      
      - name: Build Wasm
        run: |
          GOOS=wasip1 GOARCH=wasm go build -ldflags="-s -w" -trimpath -o plugin.wasm
      
      - name: Install oras
        run: |
          curl -LO https://github.com/oras-project/oras/releases/download/v1.0.0/oras_1.0.0_linux_amd64.tar.gz
          tar xzf oras_1.0.0_linux_amd64.tar.gz
          sudo mv oras /usr/local/bin/
      
      - name: Login to Registry
        run: oras login ${{ secrets.REGISTRY }} -u ${{ secrets.REGISTRY_USER }} -p ${{ secrets.REGISTRY_PASSWORD }}
      
      - name: Push Wasm
        run: |
          VERSION=${GITHUB_REF#refs/tags/}
          oras push ${{ secrets.REGISTRY }}/plugins/hello-plugin:${VERSION} \
            --artifact-type application/vnd.module.wasm.content.layer.v1+wasm \
```

## 常见问题

### Q1: 编译报错 "unsupported GOOS/GOARCH"

```bash
# 确认 Go 版本
go version  # 需要 go1.24+

# 正确编译命令
GOOS=wasip1 GOARCH=wasm go build -o plugin.wasm

# 如果还是报错，检查是否有 cgo 依赖
CGO_ENABLED=0 GOOS=wasip1 GOARCH=wasm go build -o plugin.wasm
```

### Q2: 插件加载失败

```bash
# 1. 检查 Envoy 日志
kubectl logs -n higress-system deploy/higress-gateway | grep wasm

# 2. 常见错误
# - "unable to initialize Wasm code": Wasm 文件损坏或格式错误

# - "plugin configuration failed": 配置解析错误

# 3. 验证 Wasm 文件
file plugin.wasm
wasm-objdump -x plugin.wasm  # 需要 wabt 工具
```

## 复习卡片

> [!TIP]
> **快速复习清单**
>
> 1. **环境**：Go 1.24+、oras、Docker、kubectl
> 2. **SDK**：`github.com/higress-group/wasm-go`，wrapper 包封装 Proxy-Wasm ABI
> 3. **核心钩子**：ProcessRequestHeaders / ProcessRequestBody / ProcessResponseHeaders
> 4. **宿主调用**：proxywasm.GetHttpRequestHeader / ReplaceHttpRequestHeader / SendHttpResponse
> 5. **返回值**：types.ActionContinue 继续，types.ActionPause 暂停，拒绝时发送本地响应
> 6. **HTTP 调用**：proxywasm.DispatchHttpCall（异步），回调中 ResumeHttpRequest()
> 7. **编译**：`GOOS=wasip1 GOARCH=wasm go build -o plugin.wasm`
> 8. **发布**：`oras push registry/plugins/name:version plugin.wasm`
> 9. **调试**：proxywasm.LogInfo / LogError，网关日志与请求验证
